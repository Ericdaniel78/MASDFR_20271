# 1. Fit de distribuciones
candidatas = {
    "Lognormal": stats.lognorm,
    "Gamma": stats.gamma,
    "Weibull": stats.weibull_min,
    "Pareto": stats.pareto
}

ajustes, filas = {}, []

def calcular_aic(dist, params, datos):
    log_lik = np.sum(dist.logpdf(datos, *params))
    return 2 * len(params) - 2 * log_lik

for nom, dist in candidatas.items():
    params = dist.fit(sev_gu, floc=0)
    ajustes[nom] = (dist, params)
    filas.append({"Distribución": nom, "AIC": round(calcular_aic(dist, params, sev_gu), 1)})

tabla_aic = pd.DataFrame(filas).sort_values("AIC").reset_index(drop=True)

np.random.seed(42)
n_sim = 10_000
num_polizas = len(df_modelo)

dist_sev_fit, params_sev_fit = ajustes[mejor_dist_nom]

S_ground_up = np.zeros(n_sim)
S_transformada = np.zeros(n_sim)

# Matriz de pesos por exposición para seleccionar pólizas
prob_polizas = df_modelo[col_exposicion].values / total_exposicion

# Parámetros para la Binomial Negativa Auxiliar que simula el modelo Quasi-Poisson
mu_tot = lambda_hat_glm * total_exposicion
p_nb = 1.0 / phi_quasipoisson
n_nb = mu_tot * p_nb / (1.0 - p_nb)

for i in range(n_sim):
    # 1. Simulación de N mediante la NB auxiliar (reproduce E[N]=μ y Var[N]=φ*μ)
    n_claims = stats.nbinom.rvs(n=n_nb, p=p_nb)
    
    if n_claims > 0:
        # 2. Selección de pólizas ponderada por exposición
        idx_polizas = np.random.choice(num_polizas, size=n_claims, replace=True, p=prob_polizas)
        
        d_sim = df_modelo[col_deducible].values[idx_polizas]
        sa_sim = df_modelo[col_sa].values[idx_polizas]
        coas_sim = df_modelo[col_coaseguro].values[idx_polizas]
        
        # 3. Simulación de severidad ground-up
        x_sim = dist_sev_fit.rvs(*params_sev_fit, size=n_claims)
        S_ground_up[i] = x_sim.sum()
        
        # 4. Transformaciones contractuales individuales
        x_neto = np.maximum(0, x_sim - d_sim) * (1.0 - coas_sim)
        x_pagado = np.minimum(x_neto, sa_sim)
        
        S_transformada[i] = x_pagado.sum()

# Cálculo de Métricas Finales
def obtener_metricas(S_array):
    e_s = S_array.mean()
    var_99 = np.percentile(S_array, 99)
    tvar_99 = S_array[S_array >= var_99].mean()
    return e_s, var_99, tvar_99

e_gu, var_gu, tvar_gu = obtener_metricas(S_ground_up)
e_tr, var_tr, tvar_tr = obtener_metricas(S_transformada)

res_final = pd.DataFrame({
    "Métrica (Cartera Agregada)": ["Prima Pura E[S]", "VaR 99%", "TVaR 99%", "Prima Pura Promedio / Póliza"],
    "Ground-Up (Daño Real)": [f"${e_gu:,.2f}", f"${var_gu:,.2f}", f"${tvar_gu:,.2f}", f"${e_gu/num_polizas:,.2f}"],

})
